cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(QO100_WebBridge VERSION 1.0.0 LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    include(CTest)
endif()
enable_language(Fortran)
find_package(Boost 1.74 REQUIRED)
find_package(nlohmann_json 3.9 REQUIRED)
find_package(Threads REQUIRED)
if(NOT TARGET qo100_sstv_rx)
    add_subdirectory(../decoder/sstv sstv)
endif()
if(NOT TARGET qo100_rtty)
    add_subdirectory(../decoder/rtty rtty)
endif()
add_executable(qo100_webbridge main.cpp bridge.cpp server.cpp mm_monitor.cpp sstv_rx.cpp sstv_tx.cpp rtty_rx.cpp tx2.cpp ft8.cpp ft_logbook.cpp)
target_include_directories(qo100_webbridge PRIVATE ${Boost_INCLUDE_DIRS} ../include)
target_compile_definitions(qo100_webbridge PRIVATE BOOST_ERROR_CODE_HEADER_ONLY
    QO100_SOFTWARE_VERSION="${PROJECT_VERSION}"
    QO100_WEB_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../web")
target_compile_options(qo100_webbridge PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(qo100_webbridge PRIVATE nlohmann_json::nlohmann_json Threads::Threads qo100_sstv_rx qo100_sstv_tx qo100_rtty qo100_audio_spectrum)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../decoder ${CMAKE_CURRENT_BINARY_DIR}/decoder)
target_include_directories(qo100_webbridge PRIVATE ../external/ft8/src)
target_link_libraries(qo100_webbridge PRIVATE ft8_core)
install(TARGETS qo100_webbridge RUNTIME DESTINATION bin)
install(DIRECTORY ../web/ DESTINATION share/qo100/web)
